4. HOST FIREWALL
--------------------------------------------------

Configure nftables:

table inet filter {
    chain input {
        type filter hook input priority filter;
        policy drop;

        ct state established,related accept
        ct state invalid drop
        iif "lo" accept
        ip protocol icmp accept
        ip6 nexthdr ipv6-icmp accept
        tcp dport 22 accept
        tcp dport 9651 accept
        udp dport 9651 accept
    }

    chain forward {
        type filter hook forward priority filter;
        policy drop;
    }

    chain output {
        type filter hook output priority filter;
        policy accept;
    }
}

Enable nftables at boot.

Do NOT modify SSH authentication settings.

Verify SSH remains accessible.

7. SYSTEMD
--------------------------------------------------

Create:

Use:

[Unit]
Description=PrimeWave AvalancheGo Fuji Testnet Node
After=network-online.target
Wants=network-online.target

[Service]
Type=simple
User=avalanche
Group=avalanche

ExecStart=/opt/avalanchego/avalanchego \
  --network-id=fuji \
  --http-host=127.0.0.1 \
  --http-port=9650 \
  --staking-port=9651 \
  --plugin-dir=/opt/avalanchego/plugins \
  --db-dir=/data/avalanchego/db

Restart=always
RestartSec=5

LimitNOFILE=32768

StandardOutput=append:/var/log/avalanchego/avalanchego.log
StandardError=append:/var/log/avalanchego/avalanchego-error.log

[Install]
